Overview

Goal: Replace the single-workspace hardcoded cron setup in server.js with a multi-tenant scheduler that loads all active workspaces from SQLite at startup and dynamically adds/removes jobs when workspaces install or uninstall via Slack OAuth.

Load-bearing boundary

oauth.js and events.js know nothing about scheduling. They fire callbacks (onInstall, onUninstall) that server.js wires to the registry. All scheduling logic lives in scheduler-registry.js and is testable in isolation.

  const dryRun = process.env.DRY_RUN === 'true';

  const onInstall = async (workspaceId) => {
    await registry.scheduleWorkspace(db, workspaceId, dryRun);
  };
  const onUninstall = (workspaceId) => {
    registry.unscheduleWorkspace(workspaceId);
  };

  const port = parseInt(process.env.PORT || '8080', 10);
  const httpServer = http.createServer(async (req, res) => {
    try {
      if (req.method === 'GET' && req.url === '/health') {
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ status: 'ok' }));
      } else if (await handleOAuthRequest(db, req, res, onInstall)) {
        // handled by oauth.js
      } else if (await handleEventsRequest(db, req, res, onUninstall)) {
        // handled by events.js
      } else {
        res.writeHead(404); res.end();
      }
    } catch (err) {
      console.error('[http] Unhandled error:', err.message);
      if (!res.headersSent) { res.writeHead(500); res.end(); }
    }
  });
  httpServer.listen(port, () => console.log(`HTTP server listening on port ${port}`));

  function shutdown() {
    console.log('Shutting down — stopping cron jobs');
    registry.stopAll();
    httpServer.close(() => { db.close(); process.exit(0); });
  }

  process.on('SIGTERM', shutdown);
  process.on('SIGINT', shutdown);
  console.log('Server started.');

Acceptance Criteria — verified by tests

1. On startup, cron jobs registered for all active workspaces from SQLite
2. New workspace install dynamically adds cron jobs (no restart)
3. app_uninstalled stops their cron jobs and marks workspace inactive in DB
4. Each workspace's jobs use their own config loaded from DB
5. A crash in workspace A's cron handler never stops workspace B's jobs
